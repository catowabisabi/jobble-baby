@router.get("/{user_id}/score-history", response_model=ScoreHistoryResponse)
async def get_score_history(request: Request, user_id: int, db: Session = Depends(get_db)):
    """Get score history for a user."""
    records = db.query(CVScoreHistory).filter(
        CVScoreHistory.user_id == user_id
    ).order_by(CVScoreHistory.recorded_at.desc()).limit(50).all()
    
    if not records:
        return ScoreHistoryResponse(history=[], total_count=0, best_score=None, improvement_trend=None)
    
    scores = [r.score for r in records]
    best = max(scores) if scores else None
    trend = "improving" if len(scores) >= 2 and scores[0] > scores[-1] else "stable"
    
    return ScoreHistoryResponse(
        history=records,
        total_count=len(records),
        best_score=best,
        improvement_trend=trend
    )


@router.get("/{user_id}/achievements", response_model=AchievementsListResponse)
async def get_achievements(request: Request, user_id: int, db: Session = Depends(get_db)):
    """Get all earned achievements for a user."""
    achievements = db.query(MilestoneAchievement).filter(
        MilestoneAchievement.user_id == user_id
    ).order_by(MilestoneAchievement.earned_at.desc()).all()
    
    enriched = []
    for a in achievements:
        tier_info = ACHIEVEMENT_TYPES.get(a.achievement_type, {"tier": "bronze", "description": a.achievement_type})
        enriched.append(AchievementResponse(
            id=a.id,
            achievement_type=a.achievement_type,
            tier=tier_info["tier"],
            description=tier_info["description"],
            earned_at=a.earned_at,
            metadata=a.metadata
        ))
    
    return AchievementsListResponse(
        achievements=enriched,
        total_count=len(enriched),
        recent_achievements=enriched[:3]
    )


@router.get("/{user_id}/milestones", response_model=MilestonesListResponse)
async def get_milestones(request: Request, user_id: int, db: Session = Depends(get_db)):
    """Get user's milestone progress and next targets."""
    earned = {a.achievement_type for a in db.query(MilestoneAchievement).filter(
        MilestoneAchievement.user_id == user_id
    ).all()}
    
    latest_score = db.query(CVScoreHistory).filter(
        CVScoreHistory.user_id == user_id
    ).order_by(CVScoreHistory.recorded_at.desc()).first()
    
    current_score = latest_score.score if latest_score else 0
    
    all_milestones = []
    for atype, meta in ACHIEVEMENT_TYPES.items():
        if atype == "FIVE_CVS_UPLOAD":
            cv_count = db.query(CV).filter(CV.user_id == user_id).count()
            progress = min(cv_count, 5)
            is_earned = atype in earned
            all_milestones.append(MilestoneResponse(
                achievement_type=atype,
                title=atype.replace("_", " ").title(),
                description=meta["description"],
                tier=meta["tier"],
                progress_current=progress,
                progress_target=5,
                hint=f"{5 - progress} more CVs to go" if not is_earned else "Achieved!",
                is_earned=is_earned
            ))
        elif atype.startswith("SCORE_MILESTONE_"):
            target = int(atype.split("_")[-1])
            progress = min(current_score, target)
            is_earned = atype in earned
            all_milestones.append(MilestoneResponse(
                achievement_type=atype,
                title=f"Score {target}+",
                description=meta["description"],
                tier=meta["tier"],
                progress_current=progress,
                progress_target=target,
                hint=f"Reach score {target} to unlock" if not is_earned else "Achieved!",
                is_earned=is_earned
            ))
        elif atype.startswith("CATEGORY_MASTER_"):
            cat = atype.split("_", 2)[-1]
            cat_score = latest_score.category_scores.get(cat, 0) if latest_score and latest_score.category_scores else 0
            progress = min(cat_score, 9)
            is_earned = atype in earned
            all_milestones.append(MilestoneResponse(
                achievement_type=atype,
                title=f"{cat.replace('_', ' ').title()} Master",
                description=meta["description"],
                tier=meta["tier"],
                progress_current=progress,
                progress_target=9,
                hint=f"Score 9+ in {cat.replace('_', ' ')}" if not is_earned else "Achieved!",
                is_earned=is_earned
            ))
        elif atype == "FIRST_CV_UPLOAD":
            cv_count = db.query(CV).filter(CV.user_id == user_id).count()
            progress = 1 if cv_count > 0 else 0
            is_earned = atype in earned
            all_milestones.append(MilestoneResponse(
                achievement_type=atype,
                title="First CV",
                description=meta["description"],
                tier=meta["tier"],
                progress_current=progress,
                progress_target=1,
                hint="Upload your first CV" if not is_earned else "Achieved!",
                is_earned=is_earned
            ))
    
    next_milestone = next((m for m in all_milestones if not m.is_earned), None)
    
    return MilestonesListResponse(
        next_milestone=next_milestone,
        all_milestones=all_milestones,
        earned_count=len([m for m in all_milestones if m.is_earned]),
        total_count=len(all_milestones)
    )
```

useEffect(() => {
    if (!user) return;
    
    const fetchGamification = async () => {
        try {
            const [achRes, msRes, histRes] = await Promise.all([
                fetch(`${API_URL}/api/v1/cvs/${user.id}/achievements`),
                fetch(`${API_URL}/api/v1/cvs/${user.id}/milestones`),
                fetch(`${API_URL}/api/v1/cvs/${user.id}/score-history`)
            ]);
            
            if (achRes.ok) {
                const achData = await achRes.json();
                setAchievements(achData.achievements || []);
            }
            if (msRes.ok) {
                const msData = await msRes.json();
                setMilestone(msData.next_milestone || null);
            }
            if (histRes.ok) {
                const histData = await histRes.json();
                setScoreHistory(histData);
            }
        } catch (e) {
            console.log('Gamification data fetch failed:', e);
        }
    };
    
    fetchGamification();
}, [user]);
```
