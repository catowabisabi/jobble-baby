import re, os

BASE = "/mnt/c/Users/enoma/Desktop/opencode-work/agent-works/jobble-baby/JobbleBaby"
KEYS_FILE = f"{BASE}/store/storage-keys.ts"

# Parse STORAGE_KEYS from storage-keys.ts
with open(KEYS_FILE) as f:
    content = f.read()

key_map = {}  # '@jobble/xxx' -> 'XXX' (constant name)
for match in re.finditer(r'^\s+(\w+):\s+"(@jobble/[^"]+)"', content, re.MULTILINE):
    const_name = match.group(1)
    key_val = match.group(2)
    key_map[key_val] = const_name

print(f"Loaded {len(key_map)} keys from storage-keys.ts")

# Find all tab files with hardcoded @jobble/ keys
tabs_dir = f"{BASE}/app/(tabs)"
hardcoded = set()
for fname in os.listdir(tabs_dir):
    if not fname.endswith('.tsx'):
        continue
    fpath = f"{tabs_dir}/{fname}"
    with open(fpath) as f:
        text = f.read()
    for key_val in key_map:
        if key_val in text:
            hardcoded.add((fname, key_val, key_map[key_val]))

print(f"\nFiles needing updates ({len(set(f[0] for f in hardcoded))} files):")
for fname, key_val, const_name in sorted(hardcoded):
    print(f"  {fname}: {key_val} -> STORAGE_KEYS.{const_name}")

# STEP 3 — Update each tab file
for fname, key_val, const_name in hardcoded:
    fpath = f"{tabs_dir}/{fname}"
    with open(fpath) as f:
        text = f.read()
    
    # Replace the key string with STORAGE_KEYS constant
    # Only replace in AsyncStorage.getItem/setItem/removeItem calls
    # Pattern: '@jobble/xxx' (with word boundary before)
    new_text = text.replace(f"'{key_val}'", f'STORAGE_KEYS.{const_name}')
    new_text = new_text.replace(f'"{key_val}"', f'STORAGE_KEYS.{const_name}')
    
    if new_text != text:
        with open(fpath, 'w') as f:
            f.write(new_text)
        print(f"Updated: {fname}")

print("\nDone updating tab files.")
PYEOF
