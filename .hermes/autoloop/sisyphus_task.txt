#!/usr/bin/env python3
"""
Google Play Console API - App Upload Script
Requires: google-api-python-client, google-auth
Install: pip install google-api-python-client google-auth
"""
import os
import sys
import argparse
from google.oauth2 import service_account
from googleapiclient.discovery import build
from googleapiclient.http import MediaFileUpload

SCOPES = ['https://www.googleapis.com/auth/androidpublisher']

def get_play_service():
    json_key = os.environ.get('PLAY_SERVICE_ACCOUNT_JSON')
    if not json_key:
        print("Error: PLAY_SERVICE_ACCOUNT_JSON not set")
        sys.exit(1)
    
    import base64, json
    key_data = base64.b64decode(json_key).decode()
    credentials = service_account.Credentials.from_service_account_info(
        json.loads(key_data), scopes=SCOPES
    )
    return build('androidpublisher', 'v3', credentials=credentials)

def upload_apk(service, package_name, apk_path):
    edit_request = service.edits().insert(body={}, packageName=package_name)
    result = edit_request.execute()
    edit_id = result['id']

    media = MediaFileUpload(apk_path, mimetype='application/vnd.android.package')
    apk_response = service.edits().edits().uploadApk(
        packageName=package_name, editId=edit_id, media_body=media
    ).execute()
    print(f"APK uploaded: {apk_response.get('versionCode')}")
    return edit_id

def submit_for_review(service, package_name, edit_id):
    track = 'internal'  # or 'alpha', 'beta', 'production'
    service.edits().tracks().update(
        editId=edit_id,
        packageName=package_name,
        body={'track': track}
    ).execute()
    service.edits().commit(editId=edit_id, packageName=package_name).execute()
    print("Submitted for internal testing")

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='Upload APK to Play Console')
    parser.add_argument('--apk', required=True, help='Path to APK file')
    parser.add_argument('--package', default='com.jobblebaby.app', help='Package name')
    args = parser.parse_args()

    service = get_play_service()
    edit_id = upload_apk(service, args.package, args.apk)
    submit_for_review(service, args.package, edit_id)
    print("Done!")
PYEOF
