STEP 2 — Create `.github/workflows/ci.yml`
This workflow runs on every PR and merge to master:

```yaml
name: CI
on:
  push:
    branches: [master]
  pull_request:
    branches: [master]

jobs:
  lint-and-typecheck:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
      - run: npm ci
      - run: npx tsc --noEmit
      - run: npx eslint .
```

STEP 3 — Create `.github/workflows/eas-build.yml`
This workflow triggers EAS builds on merge to master (when credentials are configured):

```yaml
name: EAS Build
on:
  push:
    branches: [master]

jobs:
  build-ios:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
      - run: npm ci
      - run: npx eas build --platform ios --local --non-interactive
        env:
          EAS_BUILD_CONTEXT: ${{ toJson(github) }}
    if: false  # Disable until EAS credentials configured

  build-android:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
      - run: npm ci
      - run: npx eas build --platform android --local --non-interactive
    if: false  # Disable until EAS credentials configured
```
